--// ENGINE LEVEL SINK //--

-- The sink function explicitly returns Sink to stop input propagation
local function SinkInput(actionName, inputState, inputObject)
return Enum.ContextActionResult.Sink
end

local function VerifyBinding()
local info = ContextActionService:GetBoundActionInfo(CONFIG.ActionName)
return info and next(info) ~= nil
end

local function DisengageHardBlock()
ContextActionService:UnbindAction(CONFIG.ActionName)
System.IsSinkBound = false
end

local function EngageHardBlock()
if System.IsSinkBound then return end -- Prevent double binding

local attempts = 0  
local maxAttempts = 3  
local success = false  

repeat  
	attempts = attempts + 1  

	-- Bind Action at High Priority to override game scripts  
	ContextActionService:BindActionAtPriority(  
		CONFIG.ActionName,  
		SinkInput,  
		false,  
		CONFIG.SinkPriority,  
		Enum.KeyCode.Two  
	)  

	-- Verification  
	if VerifyBinding() then  
		success = true  
		System.IsSinkBound = true  
	else  
		ContextActionService:UnbindAction(CONFIG.ActionName)  
		task.wait(0.05)  
	end  
until success or attempts >= maxAttempts  

if not success then  
	warn("[BF] Critical: Failed to bind input sink after retries. Resetting.")  
	-- Fallback: If we can't lock, we must reset to prevent state desync  
	return false   
end  
return true

end

--// LOGIC //--

local function UpdateVisuals()
-- Toggle Status
if System.Enabled then
ToggleLabel.Text = "LOCKING: ON [L]"
ToggleLabel.TextColor3 = Color3.fromRGB(100, 255, 100)
else
ToggleLabel.Text = "LOCKING: OFF [L]"
ToggleLabel.TextColor3 = Color3.fromRGB(150, 150, 150)
end

if not System.Enabled then  
	-- Disabled visual state  
	CounterLabel.Text = "- / -"  
	TimerLabel.Text = "DISABLED"  
	BarFill.Size = UDim2.new(0,0,1,0)  
	UIStroke.Color = Color3.fromRGB(45, 45, 45)  
	LockStatus.TextTransparency = 1  
	return  
end  

CounterLabel.Text = string.format("%d / %d", System.Count, CONFIG.MaxCount)  

if System.State == STATES.IDLE then  
	TimerLabel.Text = "IDLE"  
	TimerLabel.TextColor3 = Color3.fromRGB(100, 100, 110)  
	BarFill.Size = UDim2.new(0, 0, 1, 0)  
	BarFill.BackgroundColor3 = Color3.fromRGB(200, 30, 30)  
	UIStroke.Color = Color3.fromRGB(45, 45, 45)  
	LockStatus.TextTransparency = 1  
elseif System.State == STATES.CHARGING then  
	TimerLabel.Text = "READY"  
	TimerLabel.TextColor3 = Color3.fromRGB(150, 150, 150)  
	LockStatus.TextTransparency = 1  
elseif System.State == STATES.LOCKED then  
	UIStroke.Color = Color3.fromRGB(180, 180, 180)  
	BarFill.BackgroundColor3 = Color3.fromRGB(180, 0, 0)  
	TimerLabel.TextColor3 = Color3.fromRGB(255, 255, 255)  
	LockStatus.TextTransparency = 0  
end

end

local function FullReset()
DisengageHardBlock()

System.State = STATES.IDLE  
System.Count = 0  
System.Timer = 0  

UpdateVisuals()

end

local function PlayEndSound()
if SoundParams and SoundParams.Parent then
SoundParams:Play()
end
end

local function TransitionToLocked()
if System.State == STATES.LOCKED then return end

System.State = STATES.LOCKED  
System.Timer = CONFIG.Duration  

-- Visual pulse start  
local tween = TweenService:Create(UIStroke, TweenInfo.new(0.2), {Color = Color3.fromRGB(255, 50, 50)})  
tween:Play()  
UpdateVisuals()  

-- DELAYED LOCK LOGIC  
-- We do NOT bind immediately. We wait lockDelay seconds.  
-- This ensures the game processes the 3rd key press.  
task.delay(CONFIG.LockDelay, function()  
	-- Verification inside the delayed thread:  
	-- 1. Must still be enabled  
	-- 2. Must still be in LOCKED state (user didn't reset)  
	if System.Enabled and System.State == STATES.LOCKED then  
		local bound = EngageHardBlock()  
		if not bound then  
			FullReset() -- Force reset if binding failed critically  
		end  
	end  
end)

end

--// TOGGLE LOGIC //--
local function ToggleSystem()
System.Enabled = not System.Enabled

if not System.Enabled then  
	-- If turning OFF, immediate cleanup  
	FullReset()  
else  
	-- If turning ON, just visual update, state starts at IDLE  
	UpdateVisuals()  
end

end

--// LOOPS & INPUT //--

RunService.RenderStepped:Connect(function(dt)
if not System.Enabled then return end

if System.State == STATES.LOCKED then  
	-- Count failsafe  
	if System.Count < CONFIG.MaxCount then  
		FullReset()  
		return  
	end  

	System.Timer = System.Timer - dt  

	-- Visuals  
	local progress = math.clamp(System.Timer / CONFIG.Duration, 0, 1)  
	BarFill.Size = UDim2.new(progress, 0, 1, 0)  
	TimerLabel.Text = string.format("COOLDOWN: %.1fs", math.max(0, System.Timer))  

	local pulse = (math.sin(tick() * 8) + 1) / 2  
	LockStatus.TextTransparency = 0.2 + (pulse * 0.5)  

	-- Timer End  
	if System.Timer <= 0 then  
		PlayEndSound()  
		FullReset()  
	end  
else  
	-- Idle/Charging bar reset  
	if BarFill.Size.X.Scale > 0 then  
		BarFill.Size = UDim2.new(0, 0, 1, 0)  
	end  
end

end)

local InputDebounce = false
local ToggleDebounce = false

UserInputService.InputBegan:Connect(function(input, gameProcessed)
-- Toggle L Handling (High Priority in logic)
if input.KeyCode == Enum.KeyCode.L and not gameProcessed then
if not ToggleDebounce then
ToggleDebounce = true
ToggleSystem()
task.wait(0.2)
ToggleDebounce = false
end
return
end

-- If Disabled or Locked, ignore inputs (Locked inputs are handled by CAS Sink)  
if not System.Enabled then return end  
if System.State == STATES.LOCKED then return end  
if gameProcessed then return end  

if input.KeyCode == Enum.KeyCode.Two then  
	if InputDebounce then return end  
	InputDebounce = true  

	System.Count = System.Count + 1  

	if System.Count > CONFIG.MaxCount then  
		System.Count = CONFIG.MaxCount  
	end  

	UpdateVisuals()  

	if System.Count >= CONFIG.MaxCount then  
		TransitionToLocked()  
	else  
		System.State = STATES.CHARGING  
	end  

	task.wait(0.05)  
	InputDebounce = false  
end

end)

LocalPlayer.CharacterAdded:Connect(FullReset)

ScreenGui.Destroying:Connect(function()
DisengageHardBlock()
end)

-- Initial State
FullReset()'
